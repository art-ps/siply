> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Нативный macOS-плеер: дерево папок слева, треки выбранной папки справа, плеер-бар снизу, воспроизведение через AVFoundation.

import Foundation

/// Трек — файл в папке. Теги могут быть не загружены: тогда title = имя файла.
struct Track: Identifiable, Hashable {
    var url: URL
    var title: String
    var artist: String = ""
    var album: String = ""
    var duration: TimeInterval = 0
    var trackNumber: Int?
    var artwork: Data?

    var id: URL { url }

    /// Только по имени файла — пока теги не прочитаны или не читаются.
    init(url: URL) {
        self.url = url
        let name = url.deletingPathExtension().lastPathComponent
        self.title = name
        self.trackNumber = Track.leadingNumber(in: name)
    }

    init(url: URL, title: String, artist: String = "", album: String = "",
         duration: TimeInterval = 0, trackNumber: Int? = nil, artwork: Data? = nil) {
        self.url = url
        self.title = title
        self.artist = artist
        self.album = album
        self.duration = duration
        self.trackNumber = trackNumber
        self.artwork = artwork
    }

    /// «03 - Song» → 3, «1-07 Song» → 7, «12. Song» → 12, «1999» → nil.
    /// Номер — 1–3 цифры с разделителем после; опционально «диск-» перед.
    static func leadingNumber(in name: String) -> Int? {
        let regex = /^\s*(?:\d{1,2}[-.])?\s*(\d{1,3})[\s.\-_)]/
        guard let match = name.firstMatch(of: regex) else { return nil }
        return Int(match.1)
    }

    /// По номеру трека (без номера — в конец), затем по имени файла как в Finder.
    static func sorted(_ tracks: [Track]) -> [Track] {
        tracks.sorted { a, b in
            switch (a.trackNumber, b.trackNumber) {
            case let (x?, y?) where x != y:
                return x < y
            case (.some, .none):
                return true
            case (.none, .some):
                return false
            default:
                return a.url.lastPathComponent.localizedStandardCompare(b.url.lastPathComponent) == .orderedAscending
            }
        }
    }

    /// «1 трек», «2 трека», «5 треков».
    static func countLabel(_ n: Int) -> String {
        let mod10 = n % 10, mod100 = n % 100
        let word: String
        if mod10 == 1 && mod100 != 11 { word = "трек" }
        else if (2...4).contains(mod10) && !(12...14).contains(mod100) { word = "трека" }
        else { word = "треков" }
        return "\(n) \(word)"
    }
}

extension TimeInterval {
    /// 83 → «1:23». NaN, 0 и отрицательные → «0:00».
    var mmss: String {
        guard isFinite, self > 0 else { return "0:00" }
        let total = Int(rounded())
        return String(format: "%d:%02d", total / 60, total % 60)
    }
}
```

import Foundation

/// Временная папка для тестов. Удаляется при деините.
final class TempDir {
    let url: URL

    init() {
        url = URL(fileURLWithPath: NSTemporaryDirectory())
            .appendingPathComponent("siply-tests-\(UUID().uuidString)", isDirectory: true)
        try! FileManager.default.createDirectory(at: url, withIntermediateDirectories: true)
    }

    deinit { try? FileManager.default.removeItem(at: url) }

    @discardableResult
    func file(_ name: String, _ bytes: Data = Data()) -> URL {
        let u = url.appendingPathComponent(name)
        try! bytes.write(to: u)
        return u
    }

    @discardableResult
    func dir(_ name: String) -> URL {
        let u = url.appendingPathComponent(name, isDirectory: true)
        try! FileManager.default.createDirectory(at: u, withIntermediateDirectories: true)
        return u
    }
}

import Foundation

/// Папка в дереве сайдбара. Дети — только подпапки, читаются с диска при первом обращении.
final class FolderNode: Identifiable, Hashable {
    let url: URL
    var id: URL { url }
    var name: String { url.lastPathComponent }

    private var cachedChildren: [FolderNode]??

    init(url: URL) {
        self.url = url.standardizedFileURL
    }

    /// nil — подпапок нет: у строки в сайдбаре не будет стрелки раскрытия.
    var children: [FolderNode]? {
        if let cached = cachedChildren { return cached }
        let subs = FolderNode.subfolders(of: url)
        let result: [FolderNode]? = subs.isEmpty ? nil : subs.map(FolderNode.init)
        cachedChildren = .some(result)
        return result
    }

    /// Видимые подпапки, отсортированные как в Finder.
    static func subfolders(of url: URL) -> [URL] {
        guard let items = try? FileManager.default.contentsOfDirectory(
            at: url, includingPropertiesForKeys: [.isDirectoryKey], options: [.skipsHiddenFiles]
        ) else {
            NSLog("Siply: не удалось прочитать папку %@", url.path)
            return []
        }
        return items
            .filter { (try? $0.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) == true }
            .sorted { $0.lastPathComponent.localizedStandardCompare($1.lastPathComponent) == .orderedAscending }
    }

    static func == (a: FolderNode, b: FolderNode) -> Bool { a.url == b.url }
    func hash(into hasher: inout Hasher) { hasher.combine(url) }
}
```

import AVFoundation

/// Чтение папки с музыкой: файлы, теги, обложка. Без состояния.
enum LibraryScanner {
    /// Только то, что AVFoundation играет сам. ALAC приходит как m4a.
    static let audioExtensions: Set<String> = ["mp3", "m4a", "aac", "flac", "wav", "aif", "aiff", "caf"]

    /// Аудиофайлы прямо в папке (подпапки не трогаем), по имени как в Finder.
    static func audioFiles(in folder: URL) -> [URL] {
        guard let items = try? FileManager.default.contentsOfDirectory(
            at: folder, includingPropertiesForKeys: [.isRegularFileKey], options: [.skipsHiddenFiles]
        ) else { return [] }
        return items
            .filter { audioExtensions.contains($0.pathExtension.lowercased()) }
            .filter { (try? $0.resourceValues(forKeys: [.isRegularFileKey]).isRegularFile) == true }
            .sorted { $0.lastPathComponent.localizedStandardCompare($1.lastPathComponent) == .orderedAscending }
    }

    /// Трек с тегами. Никогда не бросает: что не прочиталось — остаётся фолбэк по имени файла.
    static func loadTrack(_ url: URL) async -> Track {
        var track = Track(url: url)
        let asset = AVURLAsset(url: url)

        if let d = try? await asset.load(.duration), d.isNumeric, d.seconds.isFinite {
            track.duration = d.seconds
        }
        if let items = try? await asset.load(.commonMetadata) {
            for item in items {
                switch item.commonKey {
                case .commonKeyTitle?:
                    if let s = try? await item.load(.stringValue), !s.isEmpty { track.title = s }
                case .commonKeyArtist?:
                    track.artist = (try? await item.load(.stringValue)) ?? ""
                case .commonKeyAlbumName?:
                    track.album = (try? await item.load(.stringValue)) ?? ""
                case .commonKeyArtwork?:
                    track.artwork = try? await item.load(.dataValue)
                default:
                    break
                }
            }
        }
        if let n = await trackNumber(from: asset) { track.trackNumber = n }
        return track
    }

    /// Номер трека из ID3 («3/12») или iTunes-атома (big-endian в байтах 2–3).
    private static func trackNumber(from asset: AVURLAsset) async -> Int? {
        guard let items = try? await asset.load(.metadata) else { return nil }
        for item in items {
            switch item.identifier {
            case .id3MetadataTrackNumber?:
                if let s = try? await item.load(.stringValue),
                   let n = Int(s.split(separator: "/").first ?? ""), n > 0 { return n }
            case .iTunesMetadataTrackNumber?:
                if let d = try? await item.load(.dataValue), d.count >= 4 {
                    let n = Int(d[d.startIndex + 2]) << 8 | Int(d[d.startIndex + 3])
                    if n > 0 { return n }
                }
            default:
                continue
            }
        }
        return nil
    }

    /// Все треки папки с тегами, параллельно, отсортированные.
    static func tracks(in folder: URL) async -> [Track] {
        let urls = audioFiles(in: folder)
        let loaded = await withTaskGroup(of: (Int, Track).self) { group in
            for (i, url) in urls.enumerated() {
                group.addTask { (i, await loadTrack(url)) }
            }
            var out = urls.map(Track.init)
            for await (i, t) in group { out[i] = t }
            return out
        }
        return Track.sorted(loaded)
    }

    /// Обложка папки: первый трек с artwork, иначе cover.jpg / cover.png / folder.jpg.
    static func folderArtwork(for folder: URL, tracks: [Track]) -> Data? {
        if let data = tracks.lazy.compactMap(\.artwork).first { return data }
        for name in ["cover.jpg", "cover.png", "folder.jpg"] {
            if let data = try? Data(contentsOf: folder.appendingPathComponent(name)) { return data }
        }
        return nil
    }
}

```swift
import Foundation
import Observation

/// Состояние воспроизведения: очередь, текущий трек, режимы. Звук — через AudioEngine.
@MainActor
@Observable
final class PlayerModel {
    enum RepeatMode: String, CaseIterable {
        case off, all, one
    }

    private(set) var queue: [Track] = []
    private(set) var currentIndex: Int?
    private(set) var isPlaying = false
    private(set) var currentTime: TimeInterval = 0

    var duration: TimeInterval { engine.duration }
    var currentTrack: Track? { currentIndex.map { queue[$0] } }

    var volume: Float {
        didSet { engine.volume = volume; defaults.set(volume, forKey: Keys.volume) }
    }
    var shuffle: Bool {
        didSet { rebuildOrder(); defaults.set(shuffle, forKey: Keys.shuffle) }
    }
    var repeatMode: RepeatMode {
        didSet { defaults.set(repeatMode.rawValue, forKey: Keys.repeatMode) }
    }

    @ObservationIgnored private let engine: AudioEngine
    @ObservationIgnored private let defaults: UserDefaults
    /// Порядок обхода индексов queue: прямой или перемешанный (текущий — первым).
    @ObservationIgnored private var order: [Int] = []
    @ObservationIgnored private var timer: Timer?

    private enum Keys {
        static let volume = "volume"
        static let shuffle = "shuffle"
        static let repeatMode = "repeatMode"
    }

    init(engine: AudioEngine, defaults: UserDefaults = .standard) {
        self.engine = engine
        self.defaults = defaults
        volume = defaults.object(forKey: Keys.volume) as? Float ?? 1
        shuffle = defaults.bool(forKey: Keys.shuffle)
        repeatMode = RepeatMode(rawValue: defaults.string(forKey: Keys.repeatMode) ?? "") ?? .off
        engine.volume = volume
        engine.onFinished = { [weak self] in self?.trackFinished() }
    }

    // MARK: - Управление

    func play(queue: [Track], startingAt index: Int) {
        guard queue.indices.contains(index) else { return }
        self.queue = queue
        currentIndex = index
        rebuildOrder()
        if load(index) {
            startPlaying()
        } else {
            step(by: 1, wrap: false)
        }
    }

    func togglePlayPause() {
        guard currentIndex != nil else { return }
        if isPlaying {
            engine.pause()
            isPlaying = false
            stopTimer()
        } else {
            startPlaying()
        }
    }

    /// Пользовательский «следующий» — на краю очереди идёт по кругу.
    func next() { step(by: 1, wrap: true) }

    /// Больше 3 секунд от начала — перемотка в начало, иначе предыдущий трек.
    func previous() {
        if engine.currentTime > 3 { seek(to: 0); return }
        step(by: -1, wrap: true)
    }

    func seek(to time: TimeInterval) {
        engine.currentTime = time
        currentTime = time
    }

    func cycleRepeat() {
        let all = RepeatMode.allCases
        repeatMode = all[(all.firstIndex(of: repeatMode)! + 1) % all.count]
    }

    // MARK: - Внутреннее

    private func trackFinished() {
        if repeatMode == .one, let i = currentIndex, load(i) {
            startPlaying()
            return
        }
        step(by: 1, wrap: repeatMode == .all)
    }

    /// Шаг по order с пропуском нечитаемых. wrap=false на краю → halt().
    private func step(by delta: Int, wrap: Bool) {
        guard let cur = currentIndex, let start = order.firstIndex(of: cur) else { return }
        var pos = start
        for _ in 0..<order.count {
            pos += delta
            if !order.indices.contains(pos) {
                guard wrap else { halt(); return }
                pos = (pos + order.count) % order.count
            }
            let idx = order[pos]
            if load(idx) {
                currentIndex = idx
                startPlaying()
                return
            }
        }
        halt()
    }

    private func load(_ idx: Int) -> Bool {
        do {
            try engine.load(queue[idx].url)
            currentTime = 0
            return true
        } catch {
            NSLog("Siply: не удалось открыть %@: %@", queue[idx].url.lastPathComponent, "\(error)")
            return false
        }
    }

    private func startPlaying() {
        engine.play()
        isPlaying = true
        startTimer()
    }

    /// Конец очереди: остаёмся на текущем треке, в начале, на паузе.
    private func halt() {
        engine.pause()
        engine.currentTime = 0
        currentTime = 0
        isPlaying = false
        stopTimer()
    }

    private func rebuildOrder() {
        let all = Array(queue.indices)
        guard shuffle, let cur = currentIndex else { order = all; return }
        order = [cur] + all.filter { $0 != cur }.shuffled()
    }

    private func startTimer() {
        stopTimer()
        timer = Timer.scheduledTimer(withTimeInterval: 0.5, repeats: true) { [weak self] _ in
            MainActor.assumeIsolated { self?.currentTime = self?.engine.currentTime ?? 0 }
        }
    }

    private func stopTimer() {
        timer?.invalidate()
        timer = nil
    }
}

import AppKit
import Observation

/// Корневая папка и выбранная папка. Корень запоминается bookmark'ом в UserDefaults.
@MainActor
@Observable
final class Library {
    private(set) var root: FolderNode?
    var selection: FolderNode?
    /// Растёт при rescan — чтобы список треков перечитался даже для той же папки.
    private(set) var generation = 0

    @ObservationIgnored private let defaults: UserDefaults
    private static let bookmarkKey = "rootBookmark"

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        restore()
    }

    func open(_ url: URL) {
        let node = FolderNode(url: url)
        root = node
        selection = node
        defaults.set(try? url.bookmarkData(), forKey: Self.bookmarkKey)
    }

    /// Пересобирает дерево от корня; выбор остаётся на той же папке.
    func rescan() {
        guard let root else { return }
        let selected = selection?.url
        self.root = FolderNode(url: root.url)
        selection = selected.map(FolderNode.init)
        generation += 1
    }

    /// Корень из bookmark'а. Нет, устарел или папки больше нет → без корня.
    func restore() {
        guard let data = defaults.data(forKey: Self.bookmarkKey) else { return }
        var stale = false
        guard let url = try? URL(resolvingBookmarkData: data, bookmarkDataIsStale: &stale),
              FileManager.default.fileExists(atPath: url.path)
        else { root = nil; selection = nil; return }
        open(url)
    }

    func openFolderPanel() {

import AppKit
import MediaPlayer
import Observation

/// Control Center / медиаклавиши: команды → PlayerModel, состояние → MPNowPlayingInfoCenter.
@MainActor
final class NowPlaying {
    private let player: PlayerModel
    private var artworkURL: URL?
    private var artwork: MPMediaItemArtwork?

    init(player: PlayerModel) {
        self.player = player
        let center = MPRemoteCommandCenter.shared()
        center.playCommand.addTarget { _ in
            Task { @MainActor in if !player.isPlaying { player.togglePlayPause() } }
            return .success
        }
        center.pauseCommand.addTarget { _ in
            Task { @MainActor in if player.isPlaying { player.togglePlayPause() } }
            return .success
        }
        center.togglePlayPauseCommand.addTarget { _ in
            Task { @MainActor in player.togglePlayPause() }
            return .success
        }
        center.nextTrackCommand.addTarget { _ in
            Task { @MainActor in player.next() }
            return .success
        }
        center.previousTrackCommand.addTarget { _ in
            Task { @MainActor in player.previous() }
            return .success
        }
        center.changePlaybackPositionCommand.addTarget { event in
            guard let event = event as? MPChangePlaybackPositionCommandEvent else { return .commandFailed }
            Task { @MainActor in player.seek(to: event.positionTime) }
            return .success
        }
        observe()
    }

    /// Перепубликует состояние при каждом изменении наблюдаемых полей плеера.
    private func observe() {
        withObservationTracking {
            publish()
        } onChange: {
            Task { @MainActor [weak self] in self?.observe() }
        }
    }

    private func publish() {
        let center = MPNowPlayingInfoCenter.default()
        guard let track = player.currentTrack else {
            center.nowPlayingInfo = nil
            center.playbackState = .stopped
            return
        }
        var info: [String: Any] = [
            MPMediaItemPropertyTitle: track.title,
            MPMediaItemPropertyArtist: track.artist,
            MPMediaItemPropertyAlbumTitle: track.album,
            MPMediaItemPropertyPlaybackDuration: player.duration,
            MPNowPlayingInfoPropertyElapsedPlaybackTime: player.currentTime,
            MPNowPlayingInfoPropertyPlaybackRate: player.isPlaying ? 1.0 : 0.0,
        ]
        if let art = artwork(for: track) { info[MPMediaItemPropertyArtwork] = art }
        center.nowPlayingInfo = info
        center.playbackState = player.isPlaying ? .playing : .paused
    }

    /// Обложку собираем один раз на трек, не каждые полсекунды.
    private func artwork(for track: Track) -> MPMediaItemArtwork? {
        if artworkURL == track.url { return artwork }
        artworkURL = track.url
        artwork = track.artwork.flatMap(NSImage.init(data:)).map { image in
            MPMediaItemArtwork(boundsSize: image.size) { _ in image }
        }
        return artwork
    }
}

# Siply

Лёгкий файловый плеер для macOS. Открываешь папку с музыкой — слева дерево папок,
справа треки выбранной папки, снизу плеер. Без библиотеки, без базы, без зависимостей.

Форматы: MP3, AAC/M4A, FLAC, ALAC, WAV, AIFF, CAF — всё, что играет AVFoundation.

## Сборка

Нужен Xcode 26 и `xcodegen` (`brew install xcodegen`).

```
xcodegen generate
xcodebuild build -scheme Siply -configuration Debug -derivedDataPath build
open build/Build/Products/Debug/Siply.app
```

Тесты: `xcodebuild test -scheme Siply -destination 'platform=macOS'`.

## Клавиши

- ⌘O — открыть папку, ⌘R — обновить дерево
- Пробел — play/pause (когда фокус не в поиске), ⌘→ / ⌘← — следующий / предыдущий
- Медиаклавиши и Control Center работают

## Ручная проверка

1. Запуск без папки → «Откройте папку с музыкой», ⌘O открывает диалог.
2. Открыть папку с альбомами → слева дерево, корень раскрыт, справа его содержимое.
3. Клик по альбому → шапка с обложкой, «N треков · M мин», таблица с тегами.
4. Двойной клик по треку → играет, в строке иконка динамика, бар показывает трек и прогресс.
5. «Воспроизвести» с начала, «Перемешать» — случайный порядок, иконка shuffle подсвечена.
6. Конец альбома при repeat off → стоп на последнем; repeat all → с первого; repeat 1 → тот же.
7. Prev после 3 сек → с начала; раньше → предыдущий.
8. Поиск фильтрует таблицу; двойной клик из отфильтрованного — очередь всё равно вся папка.
9. Перезапуск → та же папка, громкость, shuffle/repeat.
10. Нечитаемый файл (переименовать .txt в .mp3) → пропускается без алерта.
11. Control Center показывает трек, медиаклавиши работают.
```
